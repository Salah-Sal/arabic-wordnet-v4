# خوارزمية المرحلة الثانية — التحليل اللغوي
# Stage 2 Algorithm — Linguistic Analysis

> هذه المرحلة تحوّل المراجع إلى **ناقد لغوي**: يفحص اللمّات، يؤلف التعريفات، يثري الحقول، ويسجّل القرارات في ملف YAML.
> **الشرط المسبق:** أكمل المرحلة الأولى أولاً. هذه الخوارزمية تقرأ من **الملف البحثي** الذي أنتجته المرحلة الأولى.

---

## الاصطلاحات — Conventions

```
لكل X في Y نفّذ          FOR EACH X in Y DO
إذا ... فـ               IF ... THEN
وإلا إذا ... فـ          ELSE IF ... THEN
وإلا                     ELSE
نهاية                    END
⟵                        تعيين (assignment)
▸ إجراء(هدف)             ورقة — إجراء نهائي ينفّذه المراجع
// تعليق                  ملاحظة توضيحية
┊                         مستوى التداخل (indentation = scope)
```

---

## المُدخلات — Inputs

```
// ١. المجموعة الترادفية (نفس مُدخل المرحلة الأولى)
مجموعة_ترادفية = {
    معرّف، لمّات، تعريف_عربي، تعريف_إنجليزي، نوع، علاقات
}

// ٢. الملف البحثي (مُخرج المرحلة الأولى — انظر WORKFLOW_STAGE1_PSEUDOCODE.md)
الملف_البحثي = {
    توثيق:           لمّة → {عدد_المعاجم، الفترات_الزمنية}
    تعريفات:         لمّة → [{مصدر، نص، مطابق، تشويش}]
    عائلة_الجذر:     {جذر، مشتقات، مرشحات}
    شواهد:           لمّة → [{نوع، نص، بالمعنى}]
    تطور_زمني:       لمّة → {تصنيف}
    مرشحات_المرادف:  [{كلمة، المصدر، ملاحظة}]
    arabterm:         [{مصطلح_عربي، مصطلح_أجنبي، المجال، تهجئة_بديلة}]
}
```

## المُخرج — Output

```
ملف YAML للمراجعة (انظر §3.3 في MANUAL_REVIEW_GUIDE.md)
```

## الخوارزمية

```
لكل مجموعة_ترادفية في قائمة_المراجعة نفّذ

    // ══════════════════════════════════════════════
    // الخطوة ١: التحقق من اللمّات
    // ══════════════════════════════════════════════

    لكل لمّة في مجموعة_ترادفية.لمّات نفّذ

        // أ. اختبار الإبدال
        إذا لمّة تصلح_بديلاً_لأخواتها في السياقات_المعتادة فـ
            مرادفة ⟵ صحيح
        وإلا
            ▸ reject_lemma(لمّة)
            انتقل_للمّة_التالية
        نهاية

        // ب. مبدأ منع الترادف التام
        ▸ add_nuance_note(لمّة، الفارق_الذي_يميزها_عن_أخواتها)   // إلزامي دائماً

        // ج. اللفظ المختص أولى
        إذا يوجد_لفظ_أخصّ_من(لمّة) للمفهوم فـ
            ▸ remove_lemma(لمّة)
            ▸ add_lemma(اللفظ_الأخصّ)
        نهاية

        // د. سياسة التشكيل
        إذا لمّة لها متجانسات_إملائية فـ
            ▸ add_diacritics(لمّة، الحركات_المميِّزة_فقط)
        نهاية

        // هـ. إنفاذ الفصحى
        إذا لمّة لهجية فـ
            ▸ remove_dialectal(لمّة)
            ▸ add_lemma(المقابل_الفصيح)
            انتقل_للمّة_التالية
        نهاية

        // و. كشف الترجمة الحرفية
        إذا لمّة تركيب_أعجمي_البنية فـ    // مثل: "مطر ثقيل" بدل "مطر غزير"
            ▸ remove_calque(لمّة)
            ▸ add_lemma(المقابل_العربي_الأصيل)
            انتقل_للمّة_التالية
        نهاية

        // القرار النهائي
        إذا المعنى مثبت_في_المعاجم عبر_العصور فـ
            ▸ confirm(لمّة)
        وإلا إذا المعنى مثبت لكن انزياح_زمني فـ
            ▸ nuance_note(لمّة، "period restriction")
        وإلا إذا الكلمة مولّدة بلا_أصل_كلاسيكي فـ
            ▸ add_enrichment(لمّة، usage="modern")
        وإلا
            ▸ reject_lemma(لمّة)
        نهاية

    نهاية  // لكل لمّة — التحقق


    // ══════════════════════════════════════════════
    // الخطوة ٢: اكتشاف لمّات مفقودة
    // ══════════════════════════════════════════════

    لكل مرشح في الملف_البحثي.مرشحات_المرادف نفّذ

        // اختبار الترادف الدائري
        تعريف_أ_يذكر_ب ⟵ (تعريف المرشح يذكر لمّات المجموعة)
        تعريف_ب_يذكر_أ ⟵ (تعريف لمّة المجموعة يذكر المرشح)
        ترادف_قوي ⟵ تعريف_أ_يذكر_ب و تعريف_ب_يذكر_أ

        // اختبار الإبدال
        إذا المرشح ينجح في اختبار_الإبدال مع اللمّات_المؤكدة فـ
            ▸ add_lemma(مرشح)
        وإلا إذا المرشح تهجئة_بديلة لِلمّة_موجودة فـ
            ▸ add_lemma(مرشح، variant=صحيح)
        وإلا إذا المرشح أخصّ_دلالياً (hyponym) فـ
            ▸ propose_new_synset(مرشح)
        وإلا
            ▸ reject_candidate(مرشح)
        نهاية

    نهاية  // لكل مرشح


    // ══════════════════════════════════════════════
    // الخطوة ٣: تدقيق التعريف
    // ══════════════════════════════════════════════

    إذا تعريف_AWN يطابق المعاجم فـ
        ▸ retain_definition()
    وإلا إذا التعريف_الكلاسيكي أوسع فـ
        ▸ nuance_note("scope"، الفرق)
    وإلا إذا تعريف_AWN أدق_علمياً فـ
        ▸ retain_definition()   // AWN أدقّ
    وإلا
        ▸ revise_definition()
    نهاية

    // تأليف التعريفات (إلزامي لكل مجموعة)
    ▸ ألّف_تعريفاً_مصطلحياً()              // ← انظر الإجراء الفرعي أدناه
    إذا المفهوم بسيط_محسوس أو مجرد أو فعل فـ
        ▸ ألّف_تعريفاً_لغوياً()            // ← انظر الإجراء الفرعي أدناه
    نهاية
    إذا المفهوم حيوان أو نبات أو أداة أو مفهوم_ثقافي_معقد فـ
        ▸ ألّف_تعريفاً_موسوعياً()
    نهاية

    // فحص الجودة — لكل تعريف مؤلَّف
    لكل تعريف في التعريفات_المؤلَّفة نفّذ
        ▸ تحقق_جودة(تعريف)                 // ← انظر الإجراء الفرعي أدناه
    نهاية


    // ══════════════════════════════════════════════
    // الخطوة ٤: فحص العلاقات
    // ══════════════════════════════════════════════

    // اختبار التعميم (hypernym)
    إذا "X هو فعلاً نوع من Y" فـ
        ▸ relations_ok()
    وإلا إذا "X نوع من Y تقنياً لكن يوجد أعم أقرب" فـ
        ▸ flag_relation(اقتراح = الأعم_الأقرب)
    وإلا
        ▸ flag_relation(خطأ_في_التسلسل)
    نهاية

    // فحوصات إضافية
    إذا لمّة.نوع = فعل فـ
        إذا الفعل يتعدى_بحرف فـ
            ▸ add_syntactic_frame(لمّة، "متعدٍ بـ " + الحرف)
        نهاية
        إذا خلط بين مصدر واسم_فاعل في نفس المجموعة فـ
            ▸ flag_split_needed()
        نهاية
    نهاية

    إذا الكلمة تتطلب فاعلاً عاقلاً/حيّاً/جماداً فـ
        ▸ add_nuance_note(لمّة، القيد_الدلالي)
    نهاية


    // ══════════════════════════════════════════════
    // الإثراء والتحقق الصرفي
    // ══════════════════════════════════════════════

    لكل لمّة في اللمّات_المؤكدة_والمضافة نفّذ

        // ── الإثراء (§2.6) ──
        ▸ أكّد_أو_صحّح(لمّة.root)
        ▸ عيّن(لمّة.usage)                // archaic / modern / common
        ▸ عيّن(لمّة.eloquence)            // eloquent / neologism / colloquial
        ▸ عيّن(لمّة.connotation)          // positive / negative / reverential / pejorative / neutral
        ▸ عيّن(لمّة.literal_figurative)   // literal / figurative

        إذا لمّة.literal_figurative = figurative فـ
            ▸ عيّن(لمّة.figurative_relation)
        نهاية

        ▸ add_nuance_note(لمّة)           // إلزامي — ما يميزها عن أخواتها

        // استخلاص من النص المعجمي
        لكل نص_معجمي في الملف_البحثي.تعريفات(لمّة) نفّذ
            إذا نص يحتوي "من المجاز" أو "يُستعار لـ" فـ
                ▸ عيّن(figurative + figurative_relation)
            نهاية
            إذا نص يحتوي نمط "فَعَلَ كذا بكذا" فـ
                ▸ add_syntactic_frame(الإطار)
            نهاية
            إذا نص يحتوي فاعل/مفعول شائع فـ
                ▸ عيّن(لمّة.typical_collocate)
            نهاية
            إذا نص يحتوي "مولَّد" أو "محدث" فـ
                ▸ عيّن(لمّة.usage = "modern")
            نهاية
            إذا نص يحتوي "عامي" أو "مبتذل" فـ
                ▸ عيّن(لمّة.eloquence = "colloquial")
            نهاية
            إذا نص يحتوي "لا يُقال إلا لـ" فـ
                ▸ أضف_إلى(لمّة.nuance_note، القيد)
            نهاية
        نهاية  // لكل نص معجمي

        ▸ عيّن(لمّة.source، المعجم_الداعم)

        // ── التحقق الصرفي (§2.7) ──
        إذا لمّة جمع_تكسير مدرج_كلمة_مستقلة فـ
            ▸ link_broken_plural(لمّة، المفرد)
        نهاية
        إذا خلط ى/ي أو ة/ه أو همزة فـ
            ▸ normalize_orthography(لمّة)
        نهاية
        إذا حرف_جر أو ضمير ملتحق فـ
            ▸ strip_clitics(لمّة)
        نهاية
        إذا الجذر_المُستنتج ≠ الشواهد فـ
            ▸ correct_root(لمّة، الجذر_الصحيح)
        نهاية
        إذا الصيغة غير_فصيحة فـ
            ▸ remove_dialectal(لمّة)
            ▸ add_lemma(المقابل_الفصيح)
        نهاية

        // ── إرشادات حسب نوع الكلمة (§2.8) ──
        إذا لمّة.نوع = اسم فـ
            إذا خلط بين مصدر واسم_ذات فـ
                ▸ flag_pos_mismatch()
            نهاية
            إذا خلط بين أبنية من نفس الجذر فـ    // مَأْل ≠ مال
                ▸ modify_lemma_form(لمّة، الصيغة_الصحيحة)
            نهاية
        وإلا إذا لمّة.نوع = فعل فـ
            ▸ عيّن(لمّة.syntactic_frame)   // لازم / متعدٍ بنفسه / متعدٍ بـ حرف
            إذا حرف_الجر يغيّر المعنى فـ     // رغب في ≠ رغب عن
                ▸ add_nuance_note(لمّة، الحرف + المعنى)
            نهاية
            إذا تضمين_نحوي فـ
                ▸ add_nuance_note(لمّة، التضمين)
            نهاية
        وإلا إذا لمّة.نوع = صفة فـ
            إذا المعنى_التراثي ≠ المعنى_المعاصر فـ
                ▸ nuance_note(لمّة، "semantic shift")
            نهاية
            إذا تركيب "صفة + جداً" فـ
                ▸ modify_lemma_form(لمّة، صيغة_المبالغة)   // فَعّال، مِفْعَال
            نهاية
        وإلا إذا لمّة.نوع = ظرف فـ
            // العربية لا تملك صنفاً مستقلاً للظرف
            ▸ flag_pos_mismatch()   // تحقق: هل هو جار+مجرور أو حال أو مفعول مطلق؟
        نهاية

    نهاية  // لكل لمّة — الإثراء والصرف


    // ══════════════════════════════════════════════
    // التسجيل: تجميع ملف YAML
    // ══════════════════════════════════════════════

    ▸ املأ(yaml.analysis)
    ▸ املأ(yaml.definition)
    ▸ املأ(yaml.authored_definitions)
    ▸ املأ(yaml.lemmas)              // لمّة + كل حقول الإثراء + morpho_check
    ▸ املأ(yaml.missing_lemmas)

    // الأمثلة
    إذا لا_شواهد_حقيقية في الملف_البحثي فـ
        ▸ add_example()
    نهاية
    ▸ املأ(yaml.examples)

    ▸ املأ(yaml.relations)

    // الملاءمة الثقافية
    إذا المفهوم لا_مقابل_مفرد في العربية فـ
        إذا يمكن التعبير بتعبير_اصطلاحي فـ
            ▸ add_phraset()
            ▸ عيّن(cultural_fit.lexical_gap_type = "phraset")
        وإلا
            ▸ flag_lexical_gap()
            ▸ عيّن(cultural_fit.lexical_gap_type = "lexical_gap")
        نهاية
    وإلا إذا المفهوم لا_صلة_بالسياق_العربي فـ
        ▸ flag_omission()
        ▸ عيّن(cultural_fit.lexical_gap_type = "omission")
    نهاية

    إذا التصنيف_الثقافي_الحالي خاطئ فـ
        ▸ reclassify_cultural()
    نهاية
    ▸ املأ(yaml.cultural_fit)

    // المتلازمات
    لكل لمّة في اللمّات_المؤكدة نفّذ
        إذا يوجد تلازم_قوي يحدد المعنى فـ
            ▸ add_collocation(لمّة، المتلازم)
        نهاية
    نهاية

    // التقييم
    ▸ قيّم(semantic_accuracy)        // 0–3
    ▸ قيّم(gloss_quality)            // 0–3
    ▸ قيّم(synonym_coherence)        // 0–2
    ▸ قيّم(completeness)             // 0–2
    ▸ قيّم(cultural_adequacy)        // direct / near_synonym / phraset / lexical_gap / omission

    // الأعلام — إذا لم تجد ما يؤكد الصحة، عَلِّم
    لكل قرار_مشكوك_فيه نفّذ
        ▸ أضف_علَم(العلَم_المناسب)
    نهاية

    // التصعيد
    إذا عدم_يقين أو تعارض_شواهد_لا_يمكن_حسمه فـ
        ▸ escalate()
    نهاية

    ▸ جمّع(yaml.actions)             // كل ▸ أعلاه أصبح سطراً في actions[]
    ▸ قيّم(yaml.overall)             // excellent / good / acceptable / poor

نهاية  // لكل مجموعة ترادفية
```

### ألّف_تعريفاً_مصطلحياً()

```
دالة ألّف_تعريفاً_مصطلحياً():

    // البنية: [الجنس القريب] + [الفصل النوعي]
    الجنس ⟵ تعريف_المجموعة_الأعم(hypernym)
    الفصل ⟵ ما_يميّز_عن_الأشقاء(co-hyponyms)
    التعريف ⟵ الجنس + الفصل

    ▸ تحقق_جودة(التعريف)
    أعد التعريف

نهاية_الدالة
```

### ألّف_تعريفاً_لغوياً()

```
دالة ألّف_تعريفاً_لغوياً():

    // اختر النمط الأنسب
    إذا يوجد مرادف مباشر في الملف_البحثي فـ
        النمط ⟵ "synonym"
    وإلا إذا التعريف_بالنقيض أوضح فـ
        النمط ⟵ "antonym"
    وإلا إذا التعريف_بالمثال أوضح فـ
        النمط ⟵ "example"
    وإلا إذا الاشتقاق يوضّح المعنى فـ
        النمط ⟵ "derivation"
    وإلا
        النمط ⟵ "context"
    نهاية

    التعريف ⟵ صِغ_حسب_النمط(النمط، الملف_البحثي)
    ▸ تحقق_جودة(التعريف)
    أعد التعريف

نهاية_الدالة
```

### تحقق_جودة(تعريف)

```
دالة تحقق_جودة(تعريف):

    // شروط الجودة الخمسة (القاسمي 2019)
    إذا التعريف غامض أو فيه ألفاظ غريبة فـ
        ▸ أعد_الصياغة(clarity)
    نهاية
    إذا فيه كلمات زائدة فـ
        ▸ احذف_الزائد(conciseness)
    نهاية
    إذا أوسع أو أضيق من المفهوم فـ
        ▸ اضبط_النطاق(equivalence)
    نهاية
    إذا يعرّف بالنفي ("ليس X") فـ
        ▸ أعد_صياغة_إيجابية(positive)
    نهاية
    إذا يحتوي لفظ المعرَّف نفسه فـ
        ▸ أزل_الدور(no_tautology)
    نهاية

    // العيوب الستة
    إذا الجنس يكرّر ما فيه ضمنياً فـ
        ▸ احذف_الحشو()
    نهاية
    إذا الجنس بعيد جداً في التسلسل فـ
        ▸ استخدم_الجنس_القريب()
    نهاية
    إذا التعريف أغرب من المعرَّف فـ
        ▸ بسّط()
    نهاية
    إذا A يُعرَّف بـ B و B بـ A فـ
        ▸ عرّف_بخصائص_مستقلة()
    نهاية
    إذا إحالات متسلسلة فـ
        ▸ عرّف_مباشرة()
    نهاية
    إذا لفظ مشترك بلا تمييز فـ
        ▸ أضف_قيداً_مميزاً()
    نهاية

نهاية_الدالة
```

## فهرس إجراءات المرحلة الثانية (▸)

### أوامر التعديل

| الإجراء | الوصف |
|---------|------|
| `▸ revise_definition()` | التعريف ركيك أو حرفي أو خاطئ البنية |
| `▸ modify_lemma_form(لمّة، صيغة)` | خطأ إملائي أو صرفي في اللمّة |
| `▸ add_diacritics(لمّة، حركات)` | متجانسات إملائية تحتاج تمييزاً |
| `▸ normalize_orthography(لمّة)` | خلط ى/ي، ة/ه، همزة |
| `▸ strip_clitics(لمّة)` | حرف جر أو ضمير ملتحق خطأً |
| `▸ correct_root(لمّة، جذر)` | الجذر المستنتج خاطئ |
| `▸ link_broken_plural(لمّة، مفرد)` | جمع تكسير مدرج كلمة مستقلة |
| `▸ correct_relation(اقتراح)` | علاقة hypernym/hyponym خاطئة |
| `▸ reclassify_cultural()` | تصنيف المحاذاة الثقافية خاطئ |

### أوامر الإضافة

| الإجراء | الوصف |
|---------|------|
| `▸ add_lemma(لمّة)` | مرادف صالح غير موجود في المجموعة |
| `▸ add_example()` | لا شواهد حقيقية متاحة |
| `▸ author_definitions()` | تأليف تعريف (مصطلحي على الأقل) |
| `▸ add_nuance_note(لمّة، نص)` | **إلزامي** — الفارق الدقيق لكل لمّة |
| `▸ add_collocation(لمّة، متلازم)` | تلازم قوي يحدد المعنى |
| `▸ add_syntactic_frame(لمّة، إطار)` | أفعال — حرف الجر يحدد المعنى |
| `▸ add_enrichment(لمّة، حقل، قيمة)` | usage / eloquence / connotation |
| `▸ add_phraset()` | فجوة معجمية تُسدّ بتعبير اصطلاحي |

### أوامر الحذف

| الإجراء | الوصف |
|---------|------|
| `▸ remove_lemma(لمّة)` | تفشل في اختبار الإبدال في كل السياقات |
| `▸ remove_dialectal(لمّة)` | شكل لهجي بدلاً من الفصحى |
| `▸ remove_calque(لمّة)` | تركيب أعجمي البنية |
| `▸ flag_omission()` | مفهوم لا صلة له بالسياق العربي |
